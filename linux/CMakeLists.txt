cmake_minimum_required(VERSION 3.24)
project(RGBKMeans LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui)
find_package(OpenMP REQUIRED)
function(kmeans_target name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(${name} PRIVATE ${OpenCV_LIBS})
    if(WIN32)
        target_link_libraries(${name} PRIVATE shell32)
        target_compile_definitions(${name} PRIVATE NOMINMAX)
    endif()
    if(MSVC)
        target_compile_options(${name} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/fp:strict>)
    else()
        target_compile_options(${name} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-ffp-contract=off>)
    endif()
endfunction()
kmeans_target(kmeans_sequential kmeans_sequential.cpp)
kmeans_target(kmeans_openmp kmeans_openmp.cpp)
target_link_libraries(kmeans_openmp PRIVATE OpenMP::OpenMP_CXX)
find_package(MPI QUIET COMPONENTS CXX)
if(MPI_CXX_FOUND)
    kmeans_target(kmeans_mpi kmeans_mpi.cpp)
    target_link_libraries(kmeans_mpi PRIVATE MPI::MPI_CXX)
else()
    message(WARNING "MPI not found: kmeans_mpi will not be built. Install OpenMPI or MS-MPI SDK/runtime.")
endif()
include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    kmeans_target(kmeans_cuda kmeans_cuda.cu)
    set_target_properties(kmeans_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON LINKER_LANGUAGE CXX)
    target_compile_options(kmeans_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
else()
    message(WARNING "CUDA Toolkit not found: kmeans_cuda will not be built.")
endif()
